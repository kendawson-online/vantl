// Simulate fetch and render
fetch('/api/timeline')
  .then(function (res) { return res.json(); })
  .then(function (data) {
    return timelineFromData('#api-tl', data.nodes || data, { mode: 'horizontal' });
  })
  .catch(function (err) {
    console.error('Failed to load data', err);
  });
